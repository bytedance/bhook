cmake_minimum_required(VERSION 3.22.1...4.0.2)
project(bytehook C)

enable_language(ASM)

option(BYTEHOOK_USE_SHADOWHOOK "Use ShadowHook for dl_init/dl_fini monitor on ARM" ON)
option(BYTEHOOK_BUILD_STATIC "Build static library" OFF)
option(BYTEHOOK_BUILD_SHARED "Build shared library" ON)

if((${ANDROID_ABI} STREQUAL "arm64-v8a") OR (${ANDROID_ABI} STREQUAL "x86_64"))
    set(ARCH_LINK_FLAGS "-Wl,-z,max-page-size=16384")
else()
    set(ARCH_LINK_FLAGS "")
endif()

file(GLOB SRC *.c)

macro(configure_bytehook_target target_name)
    target_compile_features(${target_name} PRIVATE c_std_11)
    target_include_directories(${target_name} PUBLIC include)
    target_include_directories(${target_name} PRIVATE . third_party/bsd third_party/lss)
    target_link_libraries(${target_name} PUBLIC log)

    if(USEASAN)
        target_compile_options(${target_name} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(${target_name} PRIVATE ${ARCH_LINK_FLAGS} -fsanitize=address)
    else()
        target_compile_options(${target_name} PRIVATE -Oz -ffunction-sections -fdata-sections)
        target_link_options(${target_name} PRIVATE ${ARCH_LINK_FLAGS} -Oz -Wl,--exclude-libs,ALL -Wl,--gc-sections -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/bytehook.map.txt)
    endif()

    if(BYTEHOOK_USE_SHADOWHOOK)
        if((${ANDROID_ABI} STREQUAL "arm64-v8a") OR (${ANDROID_ABI} STREQUAL "armeabi-v7a"))
            find_package(shadowhook REQUIRED CONFIG)
            target_link_libraries(${target_name} PRIVATE shadowhook::shadowhook)
        endif()
    else()
        target_compile_definitions(${target_name} PRIVATE BH_NO_SHADOWHOOK=1)
    endif()
endmacro()

if(BYTEHOOK_BUILD_STATIC)
    add_library(bytehook_static STATIC ${SRC})
    configure_bytehook_target(bytehook_static)
    target_compile_options(bytehook_static PRIVATE -fvisibility=hidden)
    set_target_properties(bytehook_static PROPERTIES OUTPUT_NAME "bytehook")
endif()

if(BYTEHOOK_BUILD_SHARED)
    add_library(bytehook SHARED ${SRC})
    configure_bytehook_target(bytehook)
    target_compile_options(bytehook PRIVATE -fvisibility=hidden)
    set_target_properties(bytehook PROPERTIES OUTPUT_NAME "bytehook")
endif()
